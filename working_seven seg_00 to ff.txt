#include <REG51.H>

sbit D1 = P3^4;
sbit D2 = P3^5;

unsigned char seg[] = {
    0x3F, 0x06, 0x5B, 0x4F,
    0x66, 0x6D, 0x7D, 0x07,
    0x7F, 0x6F, 0x77, 0x7C,
    0x39, 0x5E, 0x79, 0x71
};

void delay(void)
{
    unsigned int i;
    for(i = 0; i < 5000; i++);
}

void display(unsigned char num)
{
    D1 = 1;
    D2 = 1;

    P2 = ~seg[num >> 4];  // Display upper nibble
    D2 = 0;
    delay();
    D2 = 1;

    P2 = ~seg[num & 0x0F]; // Display lower nibble
    D1 = 0;
    delay();
    D1 = 1;
}

void main(void)
{
    unsigned char count;

    while(1)
    {
        for(count = 0x00; ; count++)
        {
            display(count);
        }
    }
}
